     1→/**
     2→ * Backstage (Бэкстейдж) — chat microservice v2.
     3→ *
     4→ * Standalone socket.io server on fixed port 3003 (Caddy gateway forwards
     5→ * requests with ?XTransformPort=3003 to this port).

     8→ *  - per-channel rooms: "ch:<slug>"

    18→ *    and moderator /-commands (ban/timeout/clear/slow/subsonly/so/delete/
    19→ *    mod/cmd/demo); command replies are system messages from "dypaGPT"
    20→ *  - moderation state in its own sqlite DB (data/chat.db, bun:sqlite):
    21→ *    bans, mods, custom commands, per-channel slow/subsonly, settings
    22→ *  - demo chat (DEMO_CHAT=1 or /demo on): fake VTuber viewers keep rooms alive
    23→ *  - REST for OBS / admin: /api/messages, /api/send (Bearer token),
    24→ *    /api/mods, /api/bans, /api/chstate; CORS * everywhere

   219→// ---------------------------------------------------------------------------
   220→// Plain HTTP routing, coexisting with socket.io
   221→//
   222→// With path "/" engine.io's request check matches every URL that starts with
   223→// "/", so /health, /api/* etc. would otherwise be swallowed by engine.io
   224→// ("Transport unknown"). We therefore re-wrap the listener chain (the exact
   225→// same wrapping technique engine.io / socket.io use internally): our
   226→// dispatcher handles its own routes first and forwards everything else —
   227→// engine.io polling / websocket traffic, URLs containing EIO= — untouched.
   228→// ---------------------------------------------------------------------------
   229→
   230→const REST_PATHS = new Set(['/health', '/announce', '/overlay'])
   231→
   232→const socketIoChain = httpServer.listeners('request').slice(0) as RequestListener[]
   233→httpServer.removeAllListeners('request')
   234→httpServer.on('request', (req, res) => {
   235→  const pathname = (req.url ?? '/').split('?')[0]
   236→  const isRest = REST_PATHS.has(pathname) || pathname.startsWith('/api/')
   237→  if (isRest) {
   238→    if (req.method === 'OPTIONS') {
   239→      // CORS preflight for browser clients (overlay, admin panel)
   240→      res.writeHead(204, {
   241→        'access-control-allow-origin': '*',
   242→        'access-control-allow-methods': 'GET, POST, OPTIONS',
   243→        'access-control-allow-headers': 'content-type, authorization',
   244→        'access-control-max-age': '86400',
   245→      })
   246→      res.end()
   247→      return
   248→    }
   249→    handleHttpRequest(req, res, pathname).catch((err: unknown) => {
   250→      const msg = err instanceof Error ? err.message : String(err)
   251→      if (!res.headersSent) {
   252→        sendJson(res, 500, { ok: false, error: msg })
   253→      } else {
   254→        res.end()
   255→      }
   256→    })
   257→    return
   258→  }
   259→  // everything else belongs to socket.io (engine.io handshake/polling etc.)
   260→  for (const listener of socketIoChain) {
   261→    listener.call(httpServer, req, res)
   262→  }
   263→})

   265→function readBody(req: IncomingMessage): Promise<string> {
   266→  return new Promise((resolve, reject) => {
   267→    const chunks: Buffer[] = []
   268→    let size = 0
   269→    req.on('data', (chunk: Buffer) => {
   270→      size += chunk.length
   271→      if (size > MAX_HTTP_BODY) {
   272→        reject(new Error('payload too large'))
   273→        req.destroy()
   274→        return
   275→      }
   276→      chunks.push(chunk)
   277→    })
   278→    req.on('end', () => resolve(Buffer.concat(chunks).toString('utf8')))
   279→    req.on('error', reject)
   280→  })
   281→}

   299→async function handleHttpRequest(req: IncomingMessage, res: ServerResponse, pathname: string): Promise<void> {
   300→  // --- GET /health ---
   301→  if (pathname === '/health') {
   302→    if (req.method !== 'GET' && req.method !== 'HEAD') {
   303→      sendJson(res, 405, { ok: false, error: 'method not allowed' })
   304→      return
   305→    }
   306→    const now = Date.now()
   307→    sendJson(res, 200, {
   308→      ok: true,
   309→      service: 'backstage-chat',
   310→      clients: io.engine.clientsCount,
   311→      emotes: (qEmoteCount.get() ?? { c: 0 }).c,
   312→      bans: (qBanCount.get(now) ?? { c: 0 }).c,
   313→      demo: demoEnabled === true,
   314→    })
   315→    return
   316→  }
   317→
   318→  // --- GET /api/emotes ---
   319→  if (pathname === '/api/emotes') {
   320→    if (req.method !== 'GET') {
   321→      sendJson(res, 405, { ok: false, error: 'method not allowed' })
   322→      return
   323→    }
   324→    const emotes = listEmotes()
   325→    sendJson(res, 200, { ok: true, emotes, count: emotes.length })
   326→    return
   327→  }
   328→
   329→  // --- GET /api/mods (channel param reserved; mods are global) ---
   330→  if (pathname === '/api/mods') {
   331→    if (req.method !== 'GET') {
   332→      sendJson(res, 405, { ok: false, error: 'method not allowed' })
   333→      return
   334→    }
   335→    sendJson(res, 200, { ok: true, mods: qModAll.all() })
   336→    return
   337→  }
   338→
   339→  // --- GET /api/bans ---
   340→  if (pathname === '/api/bans') {
   341→    if (req.method !== 'GET') {
   342→      sendJson(res, 405, { ok: false, error: 'method not allowed' })
   343→      return
   344→    }
   345→    sendJson(res, 200, { ok: true, bans: qBanAll.all(Date.now()) })
   346→    return
   347→  }
   348→
   349→  // --- GET /api/chstate?channel=slug ---
   350→  if (pathname === '/api/chstate') {
   351→    if (req.method !== 'GET') {
   352→      sendJson(res, 405, { ok: false, error: 'method not allowed' })
   353→      return
   354→    }
   355→    const channel = requestParams(req).get('channel') ?? ''
   356→    const st = getChstate(channel)
   357→    sendJson(res, 200, {
   358→      ok: true,
   359→      slow: st.slowSec,
   360→      subsonly: st.subsonly === 1,
   361→      demo: demoEnabled === true,
   362→    })
   363→    return
   364→  }
   365→
   366→  // --- GET /api/messages?channel=slug&limit=100&after=ms ---
   367→  if (pathname === '/api/messages') {
   368→    if (req.method !== 'GET') {
   369→      sendJson(res, 405, { ok: false, error: 'method not allowed' })
   370→      return
   371→    }
   372→    const params = requestParams(req)
   373→    const channel = params.get('channel')
   374→    if (typeof channel !== 'string' || channel.length === 0) {
   375→      sendJson(res, 400, { ok: false, error: 'bad_request', message: 'channel is required' })
   376→      return
   377→    }
   378→    const afterRaw = Number(params.get('after') ?? '0')
   379→    const after = Number.isFinite(afterRaw) && afterRaw > 0 ? afterRaw : 0
   380→    const limitRaw = Number(params.get('limit') ?? '100')
   381→    const limit = Number.isFinite(limitRaw) ? Math.min(500, Math.max(1, Math.floor(limitRaw))) : 100
   382→    const messages = (history.get(channel) ?? [])
   383→      .filter((m) => !m.deleted && (after === 0 || m.ts > after))
   384→      .slice(-limit)
   385→    sendJson(res, 200, { ok: true, messages })
   386→    return
   387→  }
   388→
   389→  // --- POST /api/send (OBS / bots): Bearer chat-token + {channelSlug, text} ---
   390→  if (pathname === '/api/send') {
   391→    if (req.method !== 'POST') {
   392→      sendJson(res, 405, { ok: false, error: 'method not allowed' })
   393→      return
   394→    }
   395→    const authHeader = req.headers.authorization
   396→    const bearer = typeof authHeader === 'string' && authHeader.startsWith('Bearer ') ? authHeader.slice(7) : null
   397→    const user = verifyToken(bearer)
   398→    if (!user) {
   399→      sendJson(res, 401, { ok: false, error: 'unauthorized', message: 'Требуется валидный Bearer-токен чата' })
   400→      return
   401→    }
   402→    let raw: string
   403→    try {
   404→      raw = await readBody(req)
   405→    } catch {
   406→      sendJson(res, 413, { ok: false, error: 'payload too large' })
   407→      return
   408→    }
   409→    let parsed: { channelSlug?: unknown; text?: unknown }
   410→    try {
   411→      parsed = JSON.parse(raw.length > 0 ? raw : '{}')
   412→    } catch {
   413→      sendJson(res, 400, { ok: false, error: 'invalid json' })
   414→      return
   415→    }
   416→    const channelSlug = parsed.channelSlug
   417→    const text = typeof parsed.text === 'string' ? parsed.text.trim() : ''
   418→    if (typeof channelSlug !== 'string' || channelSlug.length === 0) {
   419→      sendJson(res, 400, { ok: false, error: 'bad_request', message: 'channelSlug и text обязательны' })
   420→      return
   421→    }
   422→    const result = await processChatText(channelSlug, user, text, null)
   423→    if (!result.ok) {
   424→      sendJson(res, result.status ?? 400, {
   425→        ok: false,
   426→        error: result.code,
   427→        code: result.code,
   428→        message: result.message,
   429→      })
   430→      return
   431→    }
   432→    if (result.msg) {
   433→      sendJson(res, 200, { ok: true, message: result.msg })
   434→      return
   435→    }
   436→    sendJson(res, 200, {
   437→      ok: true,
   438→      command: true,
   439→      ...(result.senderOnly ? { response: result.senderOnly.text } : {}),
   440→    })
   441→    return
   442→  }
   443→
   444→  // --- GET /overlay (OBS browser source; read-only, no token) ---
   445→  if (pathname === '/overlay') {
   446→    if (req.method !== 'GET') {
   447→      sendJson(res, 405, { ok: false, error: 'method not allowed' })
   448→      return
   449→    }
   450→    const params = requestParams(req)
   451→    const channel = (params.get('channel') ?? 'dypagpt').replace(/[^a-zA-Z0-9_-]/g, '').slice(0, 64) || 'dypagpt'
   452→    const scaleRaw = Number(params.get('scale') ?? '1')
   453→    const scale = Math.min(4, Math.max(0.5, Number.isFinite(scaleRaw) && scaleRaw > 0 ? scaleRaw : 1))
   454→    const html = OVERLAY_HTML.replace('__CHANNEL_JSON__', JSON.stringify(channel)).replace(
   455→      '__SCALE_JSON__',
   456→      String(scale),
   457→    )
   458→    res.writeHead(200, {
   459→      'content-type': 'text/html; charset=utf-8',
   460→      'access-control-allow-origin': '*',
   461→    })
   462→    res.end(html)
   463→    return
   464→  }
   465→
   466→  // --- POST /announce (system message from the main app) ---
   467→  if (req.method !== 'POST') {
   468→    sendJson(res, 405, { ok: false, error: 'method not allowed' })
   469→    return
   470→  }
   471→  let raw: string
   472→  try {
   473→    raw = await readBody(req)
   474→  } catch {
   475→    sendJson(res, 413, { ok: false, error: 'payload too large' })
   476→    return
   477→  }
   478→  let parsed: { secret?: unknown; channelSlug?: unknown; text?: unknown; ts?: unknown }
   479→  try {
   480→    parsed = JSON.parse(raw.length > 0 ? raw : '{}')
   481→  } catch {
   482→    sendJson(res, 400, { ok: false, error: 'invalid json' })
   483→    return
   484→  }
   485→  if (parsed.secret !== INTERNAL_SECRET) {
   486→    sendJson(res, 401, { ok: false, error: 'unauthorized' })
   487→    return
   488→  }
   489→  const channelSlug = parsed.channelSlug
   490→  const text = typeof parsed.text === 'string' ? parsed.text.trim() : ''
   491→  if (typeof channelSlug !== 'string' || channelSlug.length === 0 || text.length === 0) {
   492→    sendJson(res, 400, { ok: false, error: 'channelSlug and text are required' })
   493→    return
   494→  }
   495→  const message: ChatMsg = {
   496→    id: randomUUID(),
   497→    channelSlug,
   498→    uid: null,
   499→    name: 'Система',
   500→    color: '#a970ff',
   501→    avatar: null,
   502→    text,
   503→    ts: typeof parsed.ts === 'number' && parsed.ts > 0 ? parsed.ts : Date.now(),
   504→    system: true,
   505→  }
   506→  appendHistory(channelSlug, message)
   507→  io.in(roomKey(channelSlug)).emit('chat:message', { message })
   508→  sendJson(res, 200, { ok: true })
   509→}

   511→// ---------------------------------------------------------------------------
   512→// Token verification (tokens are issued & signed by the main app)
   513→// token = base64url(payloadJson) + "." + base64url(hmacSha256(payloadJson, secret))
   514→// ---------------------------------------------------------------------------
   515→
   516→function b64urlDecode(input: string): Buffer {
   517→  let normalized = input.replace(/-/g, '+').replace(/_/g, '/')
   518→  while (normalized.length % 4 !== 0) normalized += '='
   519→  return Buffer.from(normalized, 'base64')
   520→}
   521→
   522→function verifyToken(token: unknown): ChatUser | null {
   523→  if (typeof token !== 'string' || token.length === 0) return null
   524→  const dot = token.indexOf('.')
   525→  if (dot <= 0 || dot !== token.lastIndexOf('.')) return null
   526→  const payloadPart = token.slice(0, dot)
   527→  const signaturePart = token.slice(dot + 1)
   528→  if (signaturePart.length === 0) return null
   529→  try {
   530→    const payloadJson = b64urlDecode(payloadPart).toString('utf8')
   531→    const expected = createHmac('sha256', CHAT_SECRET).update(payloadJson).digest()
   532→    const received = b64urlDecode(signaturePart)
   533→    if (received.length !== expected.length) return null
   534→    if (!timingSafeEqual(received, expected)) return null
   535→    const payload = JSON.parse(payloadJson)
   536→    if (payload === null || typeof payload !== 'object' || Array.isArray(payload)) return null
   537→    if (typeof payload.uid !== 'string' || payload.uid.length === 0) return null
   538→    if (typeof payload.name !== 'string') return null
   539→    if (typeof payload.exp !== 'number' || payload.exp < Date.now()) return null
   540→    const role: Role =
   541→      payload.role === 'OWNER' || payload.role === 'MOD' || payload.role === 'VIEWER' ? payload.role : 'VIEWER'
   542→    return {
   543→      uid: payload.uid,
   544→      name: payload.name,
   545→      color: typeof payload.color === 'string' && payload.color.length > 0 ? payload.color : '#a970ff',
   546→      avatar: typeof payload.avatar === 'string' && payload.avatar.length > 0 ? payload.avatar : null,
   547→      slug: typeof payload.slug === 'string' ? payload.slug : '',
   548→      sub: payload.sub === true,
   549→      adm: payload.adm === true,
   550→      role,
   551→      exp: payload.exp,
   552→    }
   553→  } catch {
   554→    return null
   555→  }
   556→}

   646→function parseDuration(input: string): number | null {
   647→  const m = /^(\d+)(s|m|h)?$/i.exec(input.trim())
   648→  if (!m) return null
   649→  const n = parseInt(m[1], 10)
   650→  if (!Number.isFinite(n) || n <= 0) return null
   651→  const suffix = (m[2] ?? '').toLowerCase()
   652→  const mult = suffix === 'm' ? 60 : suffix === 'h' ? 3600 : 1
   653→  return Math.min(2_592_000, n * mult) // seconds, capped at 30 days
   654→}

   656→function formatDuration(sec: number): string {
   657→  if (sec >= 3600 && sec % 3600 === 0) return `${sec / 3600}ч`
   658→  if (sec >= 3600) return `${Math.floor(sec / 3600)}ч ${Math.floor((sec % 3600) / 60)}м`
   659→  if (sec >= 60) return `${Math.floor(sec / 60)}м`
   660→  return `${sec}с`
   661→}

   682→function pick7tvFile(files: Array<{ name?: unknown }>): string | null {
   683→  const names = files.map((f) => (typeof f.name === 'string' ? f.name : '')).filter((n) => n.length > 0)
   684→  if (names.length === 0) return null
   685→  const webp = names.filter((n) => n.endsWith('.webp'))
   686→  const pool = webp.length > 0 ? webp : names
   687→  for (const pref of ['2x.webp', '3x.webp', '4x.webp', '1x.webp']) {
   688→    if (pool.includes(pref)) return pref
   689→  }
   690→  return pool[0]

   693→async function fetchEmotes(provider: string): Promise<EmoteRow[]> {
   694→  const rows: EmoteRow[] = []
   695→  if (provider === '7tv') {
   696→    const data = (await fetchJson(SEVEN_TV_URL, EMOTE_FETCH_TIMEOUT_MS)) as {
   697→      emotes?: Array<{ name?: unknown; data?: { host?: { url?: unknown; files?: Array<{ name?: unknown }> } } }>
   698→    }
   699→    for (const e of data.emotes ?? []) {
   700→      const name = typeof e.name === 'string' ? e.name : null
   701→      const hostUrl = typeof e.data?.host?.url === 'string' ? e.data.host.url : null
   702→      if (!name || !hostUrl) continue
   703→      const file = pick7tvFile(Array.isArray(e.data?.host?.files) ? (e.data!.host!.files as Array<{ name?: unknown }>) : [])
   704→      if (!file) continue
   705→      rows.push({ provider: '7tv', name, url: `https:${hostUrl.replace(/\/+$/, '')}/${file}` })
   706→    }
   707→  } else if (provider === 'bttv') {
   708→    const data = (await fetchJson(BTTV_URL, EMOTE_FETCH_TIMEOUT_MS)) as Array<{ id?: unknown; code?: unknown }>
   709→    for (const e of Array.isArray(data) ? data : []) {
   710→      if (typeof e.code === 'string' && e.code.length > 0 && typeof e.id === 'string' && e.id.length > 0) {
   711→        rows.push({ provider: 'bttv', name: e.code, url: `https://cdn.betterttv.net/emote/${e.id}/2x` })
   712→      }
   713→    }
   714→  } else if (provider === 'ffz') {
   715→    const data = (await fetchJson(FFZ_URL, EMOTE_FETCH_TIMEOUT_MS)) as {
   716→      sets?: Record<string, { emoticons?: Array<{ name?: unknown; urls?: Record<string, unknown> }> }>
   717→    }
   718→    for (const key of Object.keys(data.sets ?? {})) {
   719→      for (const e of data.sets[key].emoticons ?? []) {
   720→        const name = typeof e.name === 'string' ? e.name : null
   721→        const urls = e.urls ?? {}
   722→        const raw =
   723→          typeof urls['2'] === 'string'
   724→            ? urls['2']
   725→            : typeof urls['4'] === 'string'
   726→              ? urls['4']
   727→              : typeof urls['1'] === 'string'
   728→                ? urls['1']
   729→                : null
   730→        if (!name || !raw) continue
   731→        const url = raw.startsWith('//') ? `https:${raw}` : raw
   732→        rows.push({ provider: 'ffz', name, url })
   733→      }
   734→    }
   735→  }
   736→  return rows
   737→}

  1130→/**
  1131→ * Command dispatcher. Returns handled=false when the text is NOT a recognized
  1132→ * command (unknown "!..." falls through as a normal chat message, exactly
  1133→ * like on Twitch). Mod commands from VIEWERs are swallowed with a
  1134→ * sender-only "forbidden" error.
  1135→ */
  1136→async function tryCommand(
  1137→  channel: string,
  1138→  user: ChatUser,
  1139→  text: string,
  1140→  socket: Socket | null,
  1141→): Promise<{ handled: boolean; result?: CmdResult }> {
  1142→  const parts = text.split(/\s+/)
  1143→  const head = parts[0]
  1144→  const args = parts.slice(1)
  1145→
  1146→  if (head.startsWith('/')) {
  1147→    const role = effectiveRole(user)
  1148→    const name = head.slice(1).trim().toLowerCase()
  1149→    if (role === 'VIEWER') {
  1150→      return {
  1151→        handled: true,
  1152→        result: cmdError('forbidden', 'Команды модератора доступны только модерам и владельцу', 403),
  1153→      }
  1154→    }
  1155→    const handler = MOD_COMMANDS[name]
  1156→    if (!handler) {
  1157→      return { handled: true, result: cmdError('unknown_command', `Неизвестная команда ${head}. Введите !команды`) }
  1158→    }
  1159→    if (OWNER_ONLY_COMMANDS.has(name) && role !== 'OWNER') {
  1160→      return { handled: true, result: cmdError('forbidden', 'Эта команда доступна только владельцу канала', 403) }
  1161→    }
  1162→    const result = await handler({ channel, user, role, args, socket })
  1163→    return { handled: true, result }
  1164→  }
  1165→
  1166→  if (head.startsWith('!')) {
  1167→    const name = head.slice(1).trim().toLowerCase()
  1168→    const handled = await tryPublicCommand(channel, user, name)
  1169→    return handled ? { handled: true } : { handled: false }
  1170→  }
  1171→
  1172→  return { handled: false }
  1173→}

  1175→// ---------------------------------------------------------------------------
  1176→// Core message pipeline (shared by socket chat:send and POST /api/send)
  1177→// ---------------------------------------------------------------------------
  1178→
  1179→interface PipelineResult {
  1180→  ok: boolean
  1181→  code?: string
  1182→  message?: string
  1183→  status?: number
  1184→  /** normal accepted message */
  1185→  msg?: ChatMsg
  1186→  /** accepted command; senderOnly carries a private reply (/cmd list) */
  1187→  command?: boolean
  1188→  senderOnly?: ChatMsg
  1189→}

  1191→async function processChatText(
  1192→  channel: string,
  1193→  user: ChatUser,
  1194→  text: string,
  1195→  socket: Socket | null,
  1196→): Promise<PipelineResult> {
  1197→  // 1. ban / active timeout
  1198→  const ban = getActiveBan(user.uid)
  1199→  if (ban) {
  1200→    return { ok: false, code: 'banned', message: banMessage(ban), status: 403 }
  1201→  }
  1202→
  1203→  // 2. text validation
  1204→  if (text.length < 1 || text.length > 500) {
  1205→    return { ok: false, code: 'bad_text', message: 'Сообщение должно содержать от 1 до 500 символов', status: 400 }
  1206→  }
  1207→
  1208→  // 3. subscriber-only gate (MOD/OWNER bypass; default on — v1 behaviour)
  1209→  const st = getChstate(channel)
  1210→  const role = effectiveRole(user)
  1211→  if (st.subsonly === 1 && user.sub !== true && role === 'VIEWER') {
  1212→    return { ok: false, code: 'subscribers_only', message: 'Чат доступен только платным подписчикам', status: 403 }
  1213→  }
  1214→
  1215→  // 4. commands — handled BEFORE the rate limit, never persisted
  1216→  if (text.startsWith('/') || text.startsWith('!')) {
  1217→    const cmd = await tryCommand(channel, user, text, socket)
  1218→    if (cmd.handled) {
  1219→      const result: PipelineResult = { ok: true, command: true }
  1220→      if (cmd.result && cmd.result.kind === 'error') {
  1221→        result.ok = false
  1222→        result.code = cmd.result.code
  1223→        result.message = cmd.result.message
  1224→        result.status = cmd.result.status
  1225→        result.command = false
  1226→      } else if (cmd.result && cmd.result.kind === 'sender') {
  1227→        result.senderOnly = buildSystemMsg(channel, cmd.result.text)
  1228→      }
  1229→      return result
  1230→    }
  1231→    // unknown "!command" — falls through as a normal message
  1232→  }
  1233→
  1234→  // 5. rate limit (slow mode raises the per-uid window)
  1235→  const minInterval = Math.max(RATE_LIMIT_MS, st.slowSec * 1000)
  1236→  const now = Date.now()
  1237→  const last = lastSentAt.get(user.uid)
  1238→  if (last !== undefined && now - last < minInterval) {
  1239→    return { ok: false, code: 'rate_limited', message: 'Слишком часто. Подождите немного', status: 429 }
  1240→  }
  1241→
  1242→  // 6. accept
  1243→  const message: ChatMsg = {
  1244→    id: randomUUID(),
  1245→    channelSlug: channel,
  1246→    uid: user.uid,
  1247→    name: user.name,
  1248→    color: user.color,
  1249→    avatar: user.avatar,
  1250→    text,
  1251→    ts: now,
  1252→    system: false,
  1253→  }
  1254→  if (role === 'OWNER' || role === 'MOD') message.role = role
  1255→  lastSentAt.set(user.uid, now)
  1256→  appendHistory(channel, message)
  1257→  io.in(roomKey(channel)).emit('chat:message', { message })
  1258→  persistMessage(channel, message) // fire-and-forget (skips demo:/system)
  1259→  return { ok: true, msg: message }
  1260→}

  1382→io.on('connection', (socket) => {
  1383→  // Invalid / expired token -> proceed as anonymous (user = null).
  1384→  socket.data.user = verifyToken((socket.handshake.auth as { token?: unknown } | undefined)?.token)
  1385→  socket.data.channel = null
  1386→
  1387→  // overlay / read-only watcher: handshake query watch=1 allows an anonymous join
  1388→  const handshakeQuery = (socket.handshake.query ?? {}) as Record<string, unknown>
  1389→  const isWatcher = handshakeQuery.watch === '1' || handshakeQuery.watch === 1
  1390→
  1391→  socket.on('channel:join', (raw: unknown) => {
  1392→    const channelSlug = (raw as { channelSlug?: unknown } | null)?.channelSlug
  1393→    if (typeof channelSlug !== 'string' || channelSlug.length === 0) return
  1394→    const user = socket.data.user as ChatUser | null
  1395→    if (!user && !isWatcher) {
  1396→      socket.emit('system:error', { code: 'auth_required', message: 'Войдите, чтобы наблюдать за чатом' })
  1397→      return
  1398→    }
  1399→    // keep one channel per socket: silently leave the previous one
  1400→    const previous = socket.data.channel as string | null
  1401→    if (previous && previous !== channelSlug) {
  1402→      socket.leave(roomKey(previous))
  1403→      socket.data.channel = null
  1404→      io.in(roomKey(previous)).emit('presence:update', { channelSlug: previous, count: roomCount(previous) })
  1405→    }
  1406→    socket.join(roomKey(channelSlug))
  1407→    socket.data.channel = channelSlug
  1408→    socket.emit('chat:history', { channelSlug, messages: visibleHistory(channelSlug) })
  1409→    io.in(roomKey(channelSlug)).emit('presence:update', { channelSlug, count: roomCount(channelSlug) })
  1410→  })
  1411→
  1412→  socket.on('channel:leave', (raw: unknown) => {
  1413→    const channelSlug = (raw as { channelSlug?: unknown } | null)?.channelSlug
  1414→    if (typeof channelSlug !== 'string' || channelSlug.length === 0) return
  1415→    if (socket.data.channel === channelSlug) socket.data.channel = null
  1416→    socket.leave(roomKey(channelSlug))
  1417→    io.in(roomKey(channelSlug)).emit('presence:update', { channelSlug, count: roomCount(channelSlug) })
  1418→  })
  1419→
  1420→  socket.on('chat:send', (raw: unknown) => {
  1421→    const payload = raw as { channelSlug?: unknown; text?: unknown } | null
  1422→    const user = socket.data.user as ChatUser | null
  1423→    if (!user) {
  1424→      socket.emit('system:error', { code: 'auth_required', message: 'Войдите через Telegram, чтобы писать в чат' })
  1425→      return
  1426→    }
  1427→    const channelSlug = payload?.channelSlug
  1428→    if (typeof channelSlug !== 'string' || channelSlug.length === 0) return
  1429→    if (socket.data.channel !== channelSlug) return // must be a room member
  1430→    const text = typeof payload?.text === 'string' ? payload.text.trim() : ''

  1432→      .then((result) => {
  1433→        if (!result.ok && result.code) {
  1434→          socket.emit('system:error', { code: result.code, message: result.message })
  1435→          return
  1436→        }
  1437→        if (result.senderOnly) {
  1438→          // private system reply (e.g. /cmd list) — sender's socket only
  1439→          socket.emit('chat:message', { message: result.senderOnly })
  1440→        }
  1441→      })
  1442→      .catch((err: unknown) => {
  1443→        const msg = err instanceof Error ? err.message : String(err)
  1444→        console.log(`[chat:send] failed: ${msg}`)
  1445→        socket.emit('system:error', { code: 'internal', message: 'Внутренняя ошибка, попробуйте ещё раз' })
  1446→      })
  1447→  })
  1448→
  1449→  socket.on('disconnect', () => {
  1450→    const channel = socket.data.channel as string | null
  1451→    if (typeof channel === 'string' && channel.length > 0) {
  1452→      socket.data.channel = null
  1453→      // defensive: socket.io already removed the room membership on disconnect
  1454→      const room = io.sockets.adapter.rooms.get(roomKey(channel))
  1455→      let count = room ? room.size : 0
  1456→      if (room && room.has(socket.id)) count = Math.max(0, count - 1)
  1457→      io.in(roomKey(channel)).emit('presence:update', { channelSlug: channel, count })
  1458→    }
