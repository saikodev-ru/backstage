     1→"use client";
     2→
     3→// Панель владельца: управление эфиром dypaGPT, OBS-интеграция, модерация чата.
     4→
     5→import { useCallback, useEffect, useState } from "react";
     6→import {
     7→  Ban as BanIcon,
     8→  Copy,
     9→  KeyRound,
    10→  Loader2,
    11→  MonitorPlay,
    12→  Play,
    13→  Plus,
    14→  Radio,
    15→  RefreshCcw,
    16→  Save,
    17→  Server,
    18→  Square,
    19→  Sword,
    20→} from "lucide-react";
    21→import {
    22→  Dialog,
    23→  DialogContent,
    24→  DialogDescription,
    25→  DialogHeader,
    26→  DialogTitle,
    27→} from "@/components/ui/dialog";
    28→import { Button } from "@/components/ui/button";
    29→import { Input } from "@/components/ui/input";
    30→import { Label } from "@/components/ui/label";
    31→import { Switch } from "@/components/ui/switch";
    32→import { Separator } from "@/components/ui/separator";
    33→import { useBackstageStore } from "@/components/backstage/store";
    34→import { toast } from "@/hooks/use-toast";
    35→import { cn } from "@/lib/utils";
    36→import type { PublicChannel } from "@/lib/types";
    37→
    38→type BanRow = { uid: string; name: string; reason: string; until: number };
    39→type ModRow = { uid: string; name: string };
    40→
    41→type Draft = {
    42→  name: string;
    43→  liveTitle: string;
    44→  liveCategory: string;
    45→  liveTags: string;
    46→  livePreviewUrl: string;
    47→};
    48→
    49→/** Состояние движка Owncast + демо-эфира (GET /api/admin/demo-stream) */
    50→type EngineState = {
    51→  running: boolean;
    52→  engine: { engine: "owncast"; reachable: boolean; online: boolean; version: string; baseUrl: string };
    53→  rtmp: { url: string; key: string };
    54→  hls: { url: string };
    55→};
    56→
    57→function draftOf(c: PublicChannel): Draft {
    58→  return {
    59→    name: c.name,
    60→    liveTitle: c.liveTitle ?? "",
    61→    liveCategory: c.liveCategory ?? "",
    62→    liveTags: c.liveTags.join(", "),
    63→    livePreviewUrl: c.livePreviewUrl ?? "",
    64→  };
    65→}
    66→
    67→export function AdminPanel() {
    68→  const { adminOpen, setAdminOpen, channels, setChannels, patchChannel, openChannel } = useBackstageStore();
    69→  const [selected, setSelected] = useState<string | null>(null);
    70→  const [drafts, setDrafts] = useState<Record<string, Draft>>({});
    71→  const [saving, setSaving] = useState(false);
    72→  const [newName, setNewName] = useState("");
    73→
    74→  // ── Модерация и OBS (REST чат-сервиса) ─────────────────────
    75→  const [bans, setBans] = useState<BanRow[]>([]);
    76→  const [mods, setMods] = useState<ModRow[]>([]);
    77→  const [demoChat, setDemoChat] = useState<boolean | null>(null);
    78→
    79→  const chatSend = useCallback(async (text: string) => {
    80→    try {
    81→      const tr = await fetch("/api/chat/token", { cache: "no-store" });
    82→      if (!tr.ok) throw new Error("нет токена");
    83→      const { token } = (await tr.json()) as { token: string };
    84→      const res = await fetch(`/api/send?XTransformPort=3003`, {
    85→        method: "POST",
    86→        headers: { "content-type": "application/json", authorization: `Bearer ${token}` },
    87→        body: JSON.stringify({ channelSlug: selected ?? "dypagpt", text }),
    88→      });
    89→      const d = (await res.json()) as { ok: boolean; error?: string; command?: boolean };
    90→      if (!res.ok || !d.ok) throw new Error(d.error ?? "ошибка");
    91→      return true;
    92→    } catch (e) {
    93→      toast({ title: "Команда не выполнена", description: e instanceof Error ? e.message : "Ошибка", variant: "destructive" });
    94→      return false;
    95→    }
    96→  }, [selected]);
    97→
    98→  const refreshModeration = useCallback(async () => {
    99→    const slug = selected ?? "dypagpt";
   100→    try {
   101→      const [b, m, s] = await Promise.all([
   102→        fetch(`/api/bans?XTransformPort=3003`, { cache: "no-store" }).then((r) => r.json() as Promise<{ bans?: BanRow[] }>),
   103→        fetch(`/api/mods?channel=${encodeURIComponent(slug)}&XTransformPort=3003`, { cache: "no-store" }).then((r) => r.json() as Promise<{ mods?: ModRow[] }>),
   104→        fetch(`/api/chstate?channel=${encodeURIComponent(slug)}&XTransformPort=3003`, { cache: "no-store" }).then((r) => r.json() as Promise<{ demo?: boolean }>),
   105→      ]);
   106→      setBans(b.bans ?? []);
   107→      setMods(m.mods ?? []);
   108→      setDemoChat(Boolean(s.demo));
   109→    } catch {
   110→      /* чат-сервис недоступен */
   111→    }
   112→  }, [selected]);
   113→
   114→  useEffect(() => {
   115→    if (adminOpen) void refreshModeration();
   116→  }, [adminOpen, refreshModeration]);
   117→
   118→  // ── Движок Owncast: статус + демо-эфир (ffmpeg → RTMP) ───────────
   119→  const [engineState, setEngineState] = useState<EngineState | null>(null);
   120→  const [demoBusy, setDemoBusy] = useState(false);
   121→
   122→  const refreshEngine = useCallback(async () => {
   123→    try {
   124→      const res = await fetch("/api/admin/demo-stream", { cache: "no-store" });
   125→      if (!res.ok) return;
   126→      setEngineState((await res.json()) as EngineState);
   127→    } catch {
   128→      /* движок не отвечает */
   129→    }
   130→  }, []);
   131→
   132→  useEffect(() => {
   133→    if (!adminOpen) return;
   134→    void refreshEngine();
   135→    const t = setInterval(() => void refreshEngine(), 10_000);
   136→    return () => clearInterval(t);
   137→  }, [adminOpen, refreshEngine]);
   138→
   139→  const toggleDemo = useCallback(
   140→    async (action: "start" | "stop") => {
   141→      setDemoBusy(true);
   142→      try {
   143→        const res = await fetch("/api/admin/demo-stream", {
   144→          method: "POST",
   145→          headers: { "content-type": "application/json" },
   146→          body: JSON.stringify({ action }),
   147→        });
   148→        const d = (await res.json()) as { ok?: boolean; message?: string };
   149→        if (!res.ok || !d.ok) throw new Error(d.message ?? "Ошибка");
   150→        toast({
   151→          title: action === "start" ? "Демо-эфир запущен" : "Демо-эфир остановлен",
   152→          description:
   153→            action === "start"
   154→              ? "ffmpeg пушит RTMP в Owncast — плеер покажет поток через несколько секунд"
   155→              : "Эфир завершён, витрина перейдёт в оффлайн",
   156→        });
   157→        await refreshEngine();
   158→        // Обновляем каналы сразу: статус эфира меняется
   159→        const ch = await fetch("/api/channels", { cache: "no-store" }).then(
   160→          (r) => r.json() as Promise<{ channels?: PublicChannel[] }>
   161→        );
   162→        if (ch.channels) setChannels(ch.channels);
   163→      } catch (e) {
   164→        toast({ title: "Ошибка демо-эфира", description: e instanceof Error ? e.message : "Ошибка", variant: "destructive" });
   165→      } finally {
   166→        setDemoBusy(false);
   167→      }
   168→    },
   169→    [refreshEngine, setChannels]
   170→  );
   171→
   172→  const copyText = useCallback((text: string, title: string) => {
   173→    void navigator.clipboard
   174→      .writeText(text)
   175→      .then(() => toast({ title }))
   176→      .catch(() => toast({ title: "Не удалось скопировать", variant: "destructive" }));
   177→  }, []);
   178→
   179→  useEffect(() => {
   180→    if (channels.length > 0) {
   181→      setSelected((prev) => prev ?? channels[0].slug);
   182→    }
   183→  }, [channels]);
   184→
   185→  useEffect(() => {
   186→    setDrafts((prev) => {
   187→      const next = { ...prev };
   188→      for (const c of channels) {
   189→        next[c.slug] = prev[c.slug] ?? draftOf(c);
   190→      }
   191→      return next;
   192→    });
   193→  }, [channels]);
   194→
   195→  const current = channels.find((c) => c.slug === selected) ?? null;
   196→  const draft = current ? (drafts[current.slug] ?? draftOf(current)) : null;
   197→
   198→  const setDraft = useCallback((patch: Partial<Draft>) => {
   199→    if (!current) return;
   200→    setDrafts((prev) => ({ ...prev, [current.slug]: { ...prev[current.slug], ...patch } }));
   201→  }, [current]);
   202→
   203→  const save = useCallback(async () => {
   204→    if (!current || !draft) return;
   205→    setSaving(true);
   206→    try {
   207→      const res = await fetch(`/api/channels/${current.slug}`, {
   208→        method: "PATCH",
   209→        headers: { "content-type": "application/json" },
   210→        body: JSON.stringify({
   211→          name: draft.name,
   212→          liveTitle: draft.liveTitle || null,
   213→          liveCategory: draft.liveCategory || null,
   214→          liveTags: draft.liveTags.split(",").map((t) => t.trim()).filter(Boolean),
   215→          livePreviewUrl: draft.livePreviewUrl || null,
   216→        }),
   217→      });
   218→      const data = (await res.json()) as { channel?: PublicChannel; error?: string; message?: string };
   219→      if (!res.ok || !data.channel) throw new Error(data.message ?? data.error ?? "Ошибка сохранения");
   220→      patchChannel(current.slug, data.channel);
   221→      toast({
   222→        title: "Изменения сохранены",
   223→        description: `Канал ${data.channel.name} обновлён`,
   224→      });
   225→    } catch (e) {
   226→      toast({ title: "Ошибка", description: e instanceof Error ? e.message : "Ошибка", variant: "destructive" });
   227→    } finally {
   228→      setSaving(false);
   229→    }
   230→  }, [current, draft, patchChannel]);
   231→
   232→  const createChannel = useCallback(async () => {
   233→    const name = newName.trim();
   234→    if (name.length < 2) return;
   235→    try {
   236→      const res = await fetch("/api/channels", {
   237→        method: "POST",
   238→        headers: { "content-type": "application/json" },
   239→        body: JSON.stringify({ name }),
   240→      });
   241→      const data = (await res.json()) as { channel?: PublicChannel; message?: string };
   242→      if (!res.ok || !data.channel) throw new Error(data.message ?? "Ошибка создания");
   243→      setChannels([...channels, data.channel]);
   244→      setSelected(data.channel.slug);
   245→      setNewName("");
   246→      toast({ title: "Канал создан", description: `Автор «${name}» добавлен на платформу` });
   247→    } catch (e) {
   248→      toast({ title: "Ошибка", description: e instanceof Error ? e.message : "Ошибка", variant: "destructive" });
   249→    }
   250→  }, [newName, channels, setChannels]);
   251→
   252→  const copyKey = () => {
   253→    if (!engineState?.rtmp.key) return;
   254→    copyText(engineState.rtmp.key, "Stream key скопирован");
   255→  };
   256→
   257→  return (
   258→    <Dialog open={adminOpen} onOpenChange={setAdminOpen}>
   259→      <DialogContent className="max-h-[90vh] max-w-2xl overflow-y-auto border-border bg-card">
   260→        <DialogHeader>
   261→          <DialogTitle className="headline flex items-center gap-2 text-base">
   262→            <Radio className="size-5 text-primary" />
   263→            ПУЛЬТ ЭФИРА
   264→          </DialogTitle>
   265→          <DialogDescription>
   266→            Движок Owncast, OBS-интеграция и модерация чата канала dypaGPT.
   267→          </DialogDescription>
   268→        </DialogHeader>
   269→
   270→        {/* Выбор канала */}
   271→        <div className="flex flex-wrap gap-2">
   272→          {channels.map((c) => (
   273→            <button
   274→              key={c.slug}
   275→              onClick={() => setSelected(c.slug)}
   276→              className={cn(
   277→                "flex items-center gap-2 rounded-md border px-3 py-1.5 text-sm transition-colors",
   278→                selected === c.slug
   279→                  ? "border-primary/60 bg-accent text-accent-foreground"
   280→                  : "border-border hover:bg-secondary"
   281→              )}
   282→            >
   283→              {c.name}
   284→              {c.isLive && <span className="size-1.5 rounded-full bg-live live-dot" aria-label="в эфире" />}
   285→            </button>
   286→          ))}
   287→        </div>
   288→
   289→        {current && draft ? (
   290→          <div className="space-y-4">
   291→            {/* Движок Owncast: реальный статус эфира + демо-эфир */}
   292→            <div className="space-y-3 rounded-lg border border-border bg-secondary/40 p-3">
   293→              <div className="flex items-center gap-2">
   294→                <Server className="size-4 text-primary" />
   295→                <span className="label-caps text-sm">СТРИМИНГ-ДВИЖОК · OWNCast</span>
   296→                {engineState && (
   297→                  <span
   298→                    className={cn(
   299→                      "label-caps ml-auto rounded-full px-2 py-px text-[10px]",
   300→                      engineState.engine.reachable
   301→                        ? "bg-[#5ad1a4]/15 text-[#5ad1a4]"
   302→                        : "bg-destructive/15 text-destructive"
   303→                    )}
   304→                  >
   305→                    {engineState.engine.reachable
   306→                      ? `ВЕРСИЯ ${engineState.engine.version || "?"}`
   307→                      : "НЕДОСТУПЕН"}
   308→                  </span>
   309→                )}
   310→              </div>
   311→
   312→              {engineState ? (
   313→                engineState.engine.reachable ? (
   314→                  <>
   315→                    <div className="flex items-center gap-2 rounded-md border border-border bg-card px-3 py-2 text-sm">
   316→                      {engineState.engine.online ? (
   317→                        <span className="live-breathe label-caps flex items-center gap-1.5 rounded-full bg-live px-2 py-px text-[11px] text-white">
   318→                          <span className="size-1.5 rounded-full bg-white live-dot" aria-hidden />
   319→                          ЭФИР ИДЁТ
   320→                        </span>
   321→                      ) : (
   322→                        <span className="label-caps rounded-full bg-secondary px-2 py-px text-[11px] text-muted-foreground">
   323→                          ОФЛАЙН
   324→                        </span>
   325→                      )}
   326→                      <span className="text-xs text-muted-foreground">
   327→                        Статус приходит из движка: эфир = RTMP-пуш из OBS
   328→                      </span>
   329→                    </div>
   330→
   331→                    {/* Демо-эфир */}
   332→                    <div className="flex items-center gap-3 rounded-md border border-border bg-card px-3 py-2">
   333→                      <Button
   334→                        variant={engineState.running ? "destructive" : "default"}
   335→                        size="sm"
   336→                        disabled={demoBusy}
   337→                        onClick={() => void toggleDemo(engineState.running ? "stop" : "start")}
   338→                        className="label-caps gap-2"
   339→                      >
   340→                        {demoBusy ? (
   341→                          <Loader2 className="size-4 animate-spin" />
   342→                        ) : engineState.running ? (
   343→                          <Square className="size-4" />
   344→                        ) : (
   345→                          <Play className="size-4" />
   346→                        )}
   347→                        {engineState.running ? "ОСТАНОВИТЬ ДЕМО-ЭФИР" : "ДЕМО-ЭФИР (FFMPEG)"}
   348→                      </Button>
   349→                      <p className="text-xs leading-snug text-muted-foreground">
   350→                        ffmpeg пушит тестовый поток в RTMP — проверка всего конвейера без OBS
   351→                      </p>
   352→                    </div>
   353→                  </>
   354→                ) : (
   355→                  <p className="rounded-md border border-destructive/30 bg-destructive/10 px-3 py-2 text-xs text-muted-foreground">
   356→                    Owncast не отвечает по адресу <code className="font-mono">{engineState.engine.baseUrl}</code>.
   357→                    Запустите движок: <code className="font-mono">cd mini-services/owncast && bun run dev</code>
   358→                  </p>
   359→                )
   360→              ) : (
   361→                <p className="text-xs text-muted-foreground">Проверяем движок…</p>
   362→              )}
   363→
   364→              {/* RTMP для OBS */}
   365→              <div className="space-y-1.5">
   366→                <Label className="label-caps text-[11px] text-muted-foreground">RTMP-СЕРВЕР ДЛЯ OBS</Label>
   367→                <div className="flex gap-2">
   368→                  <Input
   369→                    readOnly
   370→                    value={engineState?.rtmp.url ?? ""}
   371→                    aria-label="RTMP-сервер"
   372→                    className="bg-[#1a1a21] font-mono text-xs"
   373→                  />
   374→                  <Button
   375→                    variant="secondary"
   376→                    size="sm"
   377→                    className="shrink-0"
   378→                    onClick={() => engineState && copyText(engineState.rtmp.url, "RTMP-адрес скопирован")}
   379→                  >
   380→                    <Copy className="size-4" />
   381→                  </Button>
   382→                </div>
   383→              </div>
   384→            </div>
   385→
   386→            <div className="grid gap-4 sm:grid-cols-2">
   387→              <div className="space-y-1.5">
   388→                <Label htmlFor="a-name">Название канала</Label>
   389→                <Input id="a-name" value={draft.name} onChange={(e) => setDraft({ name: e.target.value })} className="bg-[#1a1a21]" />
   390→              </div>
   391→              <div className="space-y-1.5">
   392→                <Label htmlFor="a-title">Заголовок эфира</Label>
   393→                <Input id="a-title" value={draft.liveTitle} onChange={(e) => setDraft({ liveTitle: e.target.value })} placeholder="О чём эфир" className="bg-[#1a1a21]" />
   394→              </div>
   395→              <div className="space-y-1.5">
   396→                <Label htmlFor="a-cat">Категория</Label>
   397→                <Input id="a-cat" value={draft.liveCategory} onChange={(e) => setDraft({ liveCategory: e.target.value })} placeholder="Например: Разработка" className="bg-[#1a1a21]" />
   398→              </div>
   399→              <div className="space-y-1.5">
   400→                <Label htmlFor="a-tags">Теги (через запятую)</Label>
   401→                <Input id="a-tags" value={draft.liveTags} onChange={(e) => setDraft({ liveTags: e.target.value })} placeholder="код, музыка" className="bg-[#1a1a21]" />
   402→              </div>
   403→              <div className="space-y-1.5 sm:col-span-2">
   404→                <Label htmlFor="a-preview">Превью эфира (URL изображения)</Label>
   405→                <Input
   406→                  id="a-preview"
   407→                  value={draft.livePreviewUrl}
   408→                  onChange={(e) => setDraft({ livePreviewUrl: e.target.value })}
   409→                  placeholder="/img/previews/saiko-live.jpg"
   410→                  className="bg-[#1a1a21] font-mono text-xs"
   411→                />
   412→              </div>
   413→            </div>
   414→
   415→            {/* Стрим-ключ Owncast */}
   416→            <div className="rounded-lg border border-border bg-secondary/40 p-3">
   417→              <div className="flex items-center gap-2 text-sm font-semibold">
   418→                <KeyRound className="size-4 text-primary" />
   419→                Стрим-ключ Owncast
   420→              </div>
   421→              <p className="mt-1 text-xs text-muted-foreground">
   422→                Вставьте в OBS: Настройки → Вещание → Ключ потока · не передавайте ключ третьим лицам
   423→              </p>
   424→              <div className="mt-2 flex gap-2">
   425→                <Input
   426→                  readOnly
   427→                  value={engineState?.rtmp.key ?? "…"}
   428→                  aria-label="Stream key"
   429→                  className="bg-[#1a1a21] font-mono text-xs"
   430→                />
   431→                <Button variant="secondary" size="sm" onClick={copyKey}>
   432→                  Копировать
   433→                </Button>
   434→                <Button variant="secondary" size="sm" onClick={() => openChannel(current.slug)}>
   435→                  Открыть канал
   436→                </Button>
   437→              </div>
   438→            </div>
   439→
   440→            {/* OBS-интеграция */}
   441→            {current && (
   442→              <div className="space-y-3 rounded-lg border border-border bg-secondary/40 p-3">
   443→                <div className="label-caps flex items-center gap-2 text-sm">
   444→                  <MonitorPlay className="size-4 text-primary" />
   445→                  ИНТЕГРАЦИЯ С OBS
   446→                </div>
   447→                <div className="space-y-1.5">
   448→                  <Label className="label-caps text-[11px] text-muted-foreground">
   449→                    ЧАТ-ОВЕРЛЕЙ ДЛЯ BROWSER SOURCE
   450→                  </Label>
   451→                  <div className="flex gap-2">
   452→                    <Input
   453→                      readOnly
   454→                      value={`${typeof window !== "undefined" ? window.location.origin : ""}/overlay?channel=${current.slug}&XTransformPort=3003`}
   455→                      aria-label="URL оверлея чата"
   456→                      className="bg-[#1a1a21] font-mono text-xs"
   457→                    />
   458→                    <Button
   459→                      variant="secondary"
   460→                      size="sm"
   461→                      className="shrink-0"
   462→                      onClick={() =>
   463→                        void navigator.clipboard
   464→                          .writeText(`${window.location.origin}/overlay?channel=${current.slug}&XTransformPort=3003`)
   465→                          .then(() => toast({ title: "URL оверлея скопирован", description: "Вставьте в OBS → Источники → Браузер" }))
   466→                          .catch(() => toast({ title: "Не удалось скопировать", variant: "destructive" }))
   467→                      }
   468→                    >
   469→                      <Copy className="size-4" />
   470→                    </Button>
   471→                  </div>
   472→                  <p className="text-xs text-muted-foreground">
   473→                    Прозрачный чат с эмодзи 7TV/BTTV/FFZ, бейджами и авто-скроллом. Параметр
   474→                    <code className="mx-1 rounded bg-black/30 px-1 font-mono text-[10px]">&scale=1.5</code>
   475→                    увеличит размер текста.
   476→                  </p>
   477→                </div>
   478→                <div className="space-y-1.5">
   479→                  <Label className="label-caps text-[11px] text-muted-foreground">HTTP API ЧАТА (ДЛЯ БОТОВ И ВИДЖЕТОВ)</Label>
